96class BoundedCooldown:
97 """Xotirasi cheklangan va eskirgan kalitlarni avtomatik tozalovchi rate limiter (monotonic va in-place)."""
98 def __init__(self, seconds: float = 5.0, max_keys: int = 50000):
99 self.seconds = seconds
100 self.max_keys = max_keys
101 self.items: Dict[Any, float] = {}
102
103 def take(self, key: Any) -> bool:
104 now = time.monotonic()
105 if len(self.items) >= self.max_keys:
106 cutoff = now - self.seconds
107 expired = [k for k, ts in self.items.items() if ts <= cutoff]
108 for k in expired:
109 del self.items[k]
110 if len(self.items) >= self.max_keys:
111 return False
112 last = self.items.get(key, 0.0)
113 if now - last < self.seconds:
114 return False
115 self.items[key] = now
116 return True
117
118 def clear(self) -> None:
119 self.items.clear()
120

139async def _telegram_call(
140 client: httpx.AsyncClient, method: str, payload: Dict[str, Any]
141) -> Dict[str, Any]:
142 """Telegram so'rovini bajaradi, rate limit (429) va xatolarni xavfsiz boshqaradi (M8 yechimi)."""
143 try:
144 response = await client.post(f"{API_URL}/{method}", json=payload, timeout=25.0)
145 try:
146 data = response.json()
147 except ValueError:
148 return {"ok": False, "error_code": response.status_code, "delivery_unknown": True}
149
150 if not isinstance(data, dict):
151 return {"ok": False, "error_code": response.status_code, "delivery_unknown": True}
152
153 if response.status_code == 200 and data.get("ok"):
154 return data
155
156 code = data.get("error_code") or response.status_code
157 params = data.get("parameters") or {}
158 if code == 429:
159 retry_after = params.get("retry_after", 1)
160 logger.warning("Telegram 429 Too Many Requests on %s: retry_after=%s", method, retry_after)
161 else:
162 logger.warning("Telegram %s xatosi: code=%s desc=%s", method, code, str(data.get("description") or "")[:200])
163
164 return {
165 "ok": False,
166 "error_code": code,
167 "description": data.get("description", ""),
168 "parameters": params,
169 "delivery_unknown": response.status_code >= 500,
170 }
171 except httpx.RequestError as exc:
172 logger.warning("Telegram %s tarmog'ida uzilish: %s", method, type(exc).__name__)
173 return {"ok": False, "error_code": 0, "delivery_unknown": True}
174 except Exception as exc:
175 logger.error("Telegram %s so'rovida kutilmagan xato: %s", method, exc)
176 return {"ok": False, "error_code": 0, "delivery_unknown": True}
